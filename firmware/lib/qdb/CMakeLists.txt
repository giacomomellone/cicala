set(core ${CMAKE_CURRENT_LIST_DIR}/../../../core)
zephyr_library_named(cicala_qdb)
zephyr_library_sources(${core}/src/qdb.cpp ${core}/src/session.cpp ${core}/src/snapshot.cpp)
zephyr_include_directories(${core}/include)
foreach(pair "MAX_QUESTIONS;512" "RECENT_RING;20" "MAX_QUESTION_BYTES;128")
  list(GET pair 0 name)
  list(GET pair 1 fallback)
  if(DEFINED CONFIG_CICALA_${name})
    zephyr_compile_definitions(CICALA_${name}=${CONFIG_CICALA_${name}})
  else()
    zephyr_compile_definitions(CICALA_${name}=${fallback})
  endif()
endforeach()
if(CONFIG_CICALA_TEXTURE)
  zephyr_compile_definitions(CICALA_TEXTURE=1)
else()
  zephyr_compile_definitions(CICALA_TEXTURE=0)
endif()
add_dependencies(cicala_qdb zephyr_generated_headers)
